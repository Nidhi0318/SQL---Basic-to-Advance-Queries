db.orders.aggregate([
  {$match:{status:"Delivered"}},

  {$unwind:"$items"},

  {$lookup:{
    from:"products",
    localField:"items.productId",
    foreignField:"productId",
    as:"product"
  }},

  {$unwind:"$product"},

  {$group:{
    _id:"$items.productId",

    productName:{$first:"$product.name"},

    quantitySold:{$sum:"$items.quantity"},

    originalSales:{$sum:{
      $multiply:[
        "$items.quantity",
        "$items.unitPrice"
      ]
    }},

    discountAmount:{$sum:{
      $multiply:[
        "$items.quantity",
        "$items.unitPrice",
        {$divide:["$items.discount",100]}
      ]
    }},

    cost:{$sum:{
      $multiply:[
        "$items.quantity",
        "$product.costPrice"
      ]
    }}
  }},

  {$set:{
    netSales:{
      $subtract:[
        "$originalSales",
        "$discountAmount"
      ]
    }
  }},

  {$set:{
    profitBeforeDiscount:{
      $subtract:[
        "$originalSales",
        "$cost"
      ]
    },

    profitAfterDiscount:{
      $subtract:[
        "$netSales",
        "$cost"
      ]
    },

    discountPercentage:{
      $multiply:[
        {
          $divide:[
            "$discountAmount",
            "$originalSales"
          ]
        },
        100
      ]
    }
  }},

  {$project:{
    _id:0,
    productId:"$_id",
    productName:1,
    quantitySold:1,
    originalSales:1,
    discountAmount:1,
    netSales:1,
    profitBeforeDiscount:1,
    profitAfterDiscount:1,
    discountPercentage:{$round:["$discountPercentage",2]}
  }}
])
