db.products.aggregate([
  {$lookup:{
    from:"orders",
    let:{pid:"$productId"},
    pipeline:[
      {$match:{
        status:"Delivered",
        $expr:{
          $in:["$$pid","$items.productId"]
        }
      }},
      {$unwind:"$items"},
      {$match:{
        $expr:{
          $eq:["$items.productId","$$pid"]
        }
      }},
      {$group:{
        _id:null,
        quantitySold:{$sum:"$items.quantity"},
        revenue:{$sum:{
          $multiply:[
            "$items.quantity",
            "$items.unitPrice"
          ]
        }}
      }}
    ],
    as:"sales"
  }},

  {$lookup:{
    from:"reviews",
    localField:"productId",
    foreignField:"productId",
    as:"reviews"
  }},

  {$set:{
    quantitySold:{
      $ifNull:[
        {$arrayElemAt:["$sales.quantitySold",0]},
        0
      ]
    },

    revenue:{
      $ifNull:[
        {$arrayElemAt:["$sales.revenue",0]},
        0
      ]
    },

    averageRating:{$avg:"$reviews.rating"},

    totalReviews:{$size:"$reviews"},

    verifiedReviews:{
      $size:{
        $filter:{
          input:"$reviews",
          as:"review",
          cond:"$$review.verifiedPurchase"
        }
      }
    }
  }},

  {$project:{
    _id:0,
    productId:1,
    productName:"$name",
    category:1,
    quantitySold:1,
    revenue:1,
    averageRating:{$round:["$averageRating",2]},
    totalReviews:1,
    verifiedReviews:1
  }}
])
