db.orders.aggregate([
  {$group:{
    _id:null,
    totalOrders:{$sum:1},
    totalQuantity:{$sum:{$sum:"$items.quantity"}},
    totalRevenue:{$sum:{
      $sum:{
        $map:{
          input:"$items",
          as:"item",
          in:{$multiply:[
            "$$item.quantity",
            "$$item.unitPrice"
          ]}
        }
      }
    }}
  }},

  {$project:{
    _id:0,
    totalOrders:1,
    totalQuantity:1,
    totalRevenue:1,
    averageOrderValue:{
      $divide:["$totalRevenue","$totalOrders"]
    }
  }}
])
{
  totalOrders: 18,
  totalQuantity: 49,
  totalRevenue: 1685000,
  averageOrderValue: 93611.11111111111
}
db.customers.aggregate([
  {$lookup:{
    from:"orders",
    localField:"customerId",
    foreignField:"customerId",
    as:"orders"
  }},

  {$project:{
    customerId:1,
    orderCount:{$size:"$orders"}
  }},

  {$group:{
    _id:null,
    totalCustomers:{$sum:1},
    activeCustomers:{
      $sum:{$cond:[
        {$gt:["$orderCount",0]},1,0
      ]}
    },
    repeatCustomers:{
      $sum:{$cond:[
        {$gte:["$orderCount",2]},1,0
      ]}
    }
  }}
])
{
  _id: null,
  totalCustomers: 10,
  activeCustomers: 10,
  repeatCustomers: 8
}
db.orders.aggregate([
  {$group:{
    _id:null,

    delivered:{
      $sum:{$cond:[
        {$eq:["$status","Delivered"]},1,0
      ]}
    },

    cancelled:{
      $sum:{$cond:[
        {$eq:["$status","Cancelled"]},1,0
      ]}
    },

    pending:{
      $sum:{$cond:[
        {$eq:["$status","Pending"]},1,0
      ]}
    },

    totalOrders:{$sum:1}
  }},

  {$project:{
    _id:0,
    delivered:1,
    cancelled:1,
    pending:1,

    cancellationRate:{
      $multiply:[
        {$divide:["$cancelled","$totalOrders"]},
        100
      ]
    }
  }}
])
