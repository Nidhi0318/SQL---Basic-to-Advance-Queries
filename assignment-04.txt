db.orders.aggregate([
  {$match:{status:"Delivered"}},

  {$unwind:"$items"},

  {$lookup:{
    from:"products",
    localField:"items.productId",
    foreignField:"productId",
    as:"product"
  }},

  {$unwind:"$product"},

  {$group:{
    _id:"$product.category",

    products:{$addToSet:"$items.productId"},

    quantitySold:{$sum:"$items.quantity"},

    revenue:{$sum:{
      $multiply:[
        "$items.quantity",
        "$items.unitPrice"
      ]
    }},

    cost:{$sum:{
      $multiply:[
        "$items.quantity",
        "$product.costPrice"
      ]
    }},

    averageSellingPrice:{$avg:"$items.unitPrice"}
  }},

  {$project:{
    _id:0,

    category:"$_id",

    uniqueProducts:{$size:"$products"},

    quantitySold:1,

    revenue:{$round:["$revenue",2]},

    averageSellingPrice:{
      $round:["$averageSellingPrice",2]
    },

    profit:{
      $round:[
        {$subtract:["$revenue","$cost"]},
        2
      ]
    },

    profitMargin:{
      $round:[
        {
          $multiply:[
            {
              $divide:[
                {$subtract:["$revenue","$cost"]},
                "$revenue"
              ]
            },
            100
          ]
        },
        2
      ]
    }
  }},

  {$sort:{profit:-1}}
])
