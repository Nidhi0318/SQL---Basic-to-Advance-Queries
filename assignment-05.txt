db.customers.aggregate([
  {$lookup:{
    from:"orders",
    localField:"customerId",
    foreignField:"customerId",
    as:"orders"
  }},

  {$unwind:"$orders"},

  {$match:{
    "orders.status":"Delivered"
  }},

  {$set:{
    orderTotal:{
      $sum:{
        $map:{
          input:"$orders.items",
          as:"item",
          in:{
            $multiply:[
              "$$item.quantity",
              "$$item.unitPrice"
            ]
          }
        }
      }
    }
  }},

  {$group:{
    _id:{
      state:"$state",
      city:"$city"
    },

    customers:{$addToSet:"$customerId"},

    orders:{$sum:1},

    totalSales:{$sum:"$orderTotal"},

    quantitySold:{$sum:{
      $sum:"$orders.items.quantity"
    }}
  }},

  {$group:{
    _id:null,

    regions:{$push:"$$ROOT"},

    companySales:{$sum:"$totalSales"}
  }},

  {$unwind:"$regions"},

  {$project:{
    _id:0,

    state:"$regions._id.state",

    city:"$regions._id.city",

    customers:{$size:"$regions.customers"},

    orders:"$regions.orders",

    totalSales:"$regions.totalSales",

    averageOrderValue:{
      $divide:[
        "$regions.totalSales",
        "$regions.orders"
      ]
    },

    quantitySold:"$regions.quantitySold",

    salesContribution:{
      $multiply:[
        {
          $divide:[
            "$regions.totalSales",
            "$companySales"
          ]
        },
        100
      ]
    }
  }},

  {$sort:{
    totalSales:-1
  }}
])
