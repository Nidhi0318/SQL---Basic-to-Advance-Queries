db.orders.aggregate([
  {$match:{status:"Delivered"}},

  {$group:{
    _id:"$customerId",
    firstOrder:{$min:"$orderDate"}
  }},

  {$project:{
    month:{$dateToString:{
      format:"%Y-%m",
      date:"$firstOrder"
    }}
  }},

  {$group:{
    _id:"$month",
    newCustomers:{$sum:1}
