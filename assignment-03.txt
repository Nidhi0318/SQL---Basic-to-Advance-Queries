db.customers.aggregate([
  {
    $lookup: {
      from: "orders",
      localField: "customerId",
      foreignField: "customerId",
      as: "orders"
    }
  },

  {
    $unwind: "$orders"
  },

  {
    $set: {
      orderTotal: {
        $sum: {
          $map: {
            input: "$orders.items",
            as: "item",
            in: {
              $multiply: [
                "$$item.quantity",
                "$$item.unitPrice"
              ]
            }
          }
        }
      }
    }
  },

  {
    $group: {
      _id: "$customerId",

      customerName: { $first: "$name" },

      membership: { $first: "$membership" },

      totalOrders: { $sum: 1 },

      totalProducts: {
        $sum: {
          $sum: "$orders.items.quantity"
        }
      },

      totalSpending: {
        $sum: "$orderTotal"
      },

      firstOrderDate: {
        $min: "$orders.orderDate"
      },

      lastOrderDate: {
        $max: "$orders.orderDate"
      }
    }
  },

  {
    $set: {
      averageOrderValue: {
        $divide: [
          "$totalSpending",
          "$totalOrders"
        ]
      }
    }
  },

  {
    $set: {
      customerSegment: {
        $switch: {
          branches: [
            {
              case: { $gte: ["$totalSpending", 500000] },
              then: "Platinum"
            },
            {
              case: { $gte: ["$totalSpending", 200000] },
              then: "Gold"
            },
            {
              case: { $gte: ["$totalSpending", 50000] },
              then: "Silver"
            }
          ],
          default: "Bronze"
        }
      }
    }
  },

  {
    $project: {
      _id: 0,
      customerId: "$_id",
      customerName: 1,
      membership: 1,
      totalOrders: 1,
      totalProducts: 1,
      totalSpending: 1,
      averageOrderValue: {
        $round: ["$averageOrderValue", 2]
      },
      firstOrderDate: 1,
      lastOrderDate: 1,
      customerSegment: 1
    }
  },

  {
    $sort: {
      totalSpending: -1
    }
  }
])
