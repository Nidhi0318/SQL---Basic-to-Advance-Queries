db.customers.aggregate([
  {$lookup:{
    from:"orders",
    localField:"customerId",
    foreignField:"customerId",
    as:"orders"
  }},

  {$set:{
    lastOrderDate:{$max:"$orders.orderDate"}
  }},

  {$match:{
    lastOrderDate:{
      $lt:new Date(
        new Date().setDate(new Date().getDate()-90)
      )
    }
  }},

  {$project:{
    _id:0,
    customerId:1,
    customerName:"$name",
    lastOrderDate:1
  }}
])
